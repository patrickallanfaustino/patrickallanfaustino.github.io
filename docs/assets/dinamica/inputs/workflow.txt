gmx pdb2gmx -v -f protonated.pdb -o xilanase.gro -ignh
gmx editconf -f xilanase_water.gro -o box.gro -c -d 1.2 -bt dodecahedron
gmx solvate -cp box.gro -cs tip4p.gro -o solvated.gro -p topol.top

gmx grompp -v -f inputs/ions.mdp -c solvated.gro -o ions.tpr -p topol.top -maxwarn 1
gmx select -s ions.tpr -on water_bulk.ndx -select '"SOL"resname SOL and not resnr 186 to 192'
gmx genion -s ions.tpr -n water_bulk.ndx -o neutralized.gro -p topol.top -pname NA -nname CL -neutral -conc 0.15

gmx make_ndx -f neutralized.gro -o index.ndx
gmx genrestr -f neutralized.gro -n index.ndx -o posre_bb.itp -fc 1000 1000 1000
Escolha 'MainChain'
gmx genrestr -f neutralized.gro -n index.ndx -o posre_sc.itp -fc 1000 1000 1000
Escolha 'SideChain-H'

awk '($1 ~ /^[0-9]+$/ && NF == 5) {printf "%6s %5s %14s %14s %14s\n", $1, $2, "POSRES_FC_BB", "POSRES_FC_BB", "POSRES_FC_BB"; next} {print}' posre_bb.itp > tmp && mv tmp posre_bb.itp

awk '($1 ~ /^[0-9]+$/ && NF == 5) {printf "%6s %5s %14s %14s %14s\n", $1, $2, "POSRES_FC_SC", "POSRES_FC_SC", "POSRES_FC_SC"; next} {print}' posre_sc.itp > tmp && mv tmp posre_sc.itp
